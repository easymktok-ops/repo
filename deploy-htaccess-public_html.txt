# ============================================================================
# .htaccess para la RAIZ del sitio  ->  sube esto como  public_html/.htaccess
# ----------------------------------------------------------------------------
# Hace 3 cosas:
#  0) Cierra la URL temporal de Webempresa (wordpresstemporal.com): todo lo que
#     llegue por ahi se redirige a tu dominio real. Asi no queda un segundo
#     sitio expuesto ni Google lo indexa como copia. Se excluye .well-known
#     para no romper la renovacion del certificado SSL.
#  1) IMPRESCINDIBLE: enruta /api/* al backend PHP (pagos, panel, CMS). Sin esta
#     regla, /api/create-checkout-session.php no existe y la reserva truena.
#  2) Canonico: www -> sin www.
# ============================================================================

<IfModule mod_rewrite.c>
  RewriteEngine On

  # 0) URL temporal de Webempresa -> dominio real (301).
  RewriteCond %{HTTP_HOST} wordpresstemporal\.com$ [NC]
  RewriteCond %{REQUEST_URI} !^/\.well-known/ [NC]
  RewriteRule ^(.*)$ https://aerodiverti.com.mx/$1 [R=301,L]

  # 1) /api/* -> backend.
  RewriteCond %{REQUEST_URI} ^/api/
  RewriteRule ^api/(.*)$ server/public/api/$1 [L]

  # 2) www -> sin www (301).
  RewriteCond %{HTTP_HOST} ^www\.aerodiverti\.com\.mx$ [NC]
  RewriteRule ^(.*)$ https://aerodiverti.com.mx/$1 [R=301,L]
</IfModule>
